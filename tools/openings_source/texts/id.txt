# Coach openings: Indonesian texts (keys of en.txt). tools/gen_opening_lang.py merges them with the names of
# families.json into assets/coach/openings/id.lang.
# Register as in the coach files: the coach says "kamu" to the player and "aku" for himself ("-mu" / "-ku").
# Indonesian has no articles, cases or gender: the "def" form of every name is its citation form, capitalised
# as a title ("Pertahanan Sisilia", "Varian Najdorf"), and fits every template unchanged.
# "Putih" / "Hitam" (the players) are capitalised. Squares as "e4", files as "lajur c", pawns as "pion d" (the
# generator spells the letter after "lajur" / "pion"); pieces in words, never piece letters. A sentence never
# starts with a bare square: "Langkah e4 ...". "light-squared bishop" = "gajah petak putih"; "unbalanced" =
# "asimetris" ("tidak seimbang" would suggest one side is better); "pin" = "mengunci" / "kuncian", so a locked
# pawn chain is "tertutup rapat", never "terkunci".
