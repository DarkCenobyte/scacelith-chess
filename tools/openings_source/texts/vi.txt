# Coach openings: Vietnamese texts (keys of en.txt). tools/gen_opening_lang.py merges them with the names of
# families.json into assets/coach/openings/vi.lang.
# Register as in the coach files (glossary/vi.md): the coach says "tôi" for himself and "bạn" to the player,
# "chúng ta" for both. Vietnamese has no articles, cases or gender: the "def" form of every name is its
# citation form, and it fits every template unchanged. Names are written in sentence case: the first syllable,
# proper names, nicknames (Rồng, Nhím, Hà mã) and piece names are capitalised ("Phòng thủ Sicilia", "Biến thể
# cổ điển", "Khai cuộc bốn Mã"); a list of several openings shares one lower-case head noun ("các phòng thủ
# Sicilia, Pháp hay Caro-Kann"). The templates avoid "khai cuộc bằng Khai cuộc ..." (the noun twice).
# Pieces are capitalised (Vua, Hậu, Xe, Tượng, Mã, Tốt), and so are the players (Trắng, Đen). "tốt" (good)
# sounds like "Tốt" (pawn): never "tốt" alone for "good" ("thuận lợi", "ổn"), only "tốt nhất".
# Squares as "e4", files as "cột c", pawns as "Tốt d" (the generator spells the letter after "cột" / "Tốt");
# pieces in words, never piece letters. A sentence never starts with a bare square: "Nước e4 ...".
# Terms: light-squared / dark-squared bishop = "Tượng ô trắng" / "Tượng ô đen"; king's / queen's bishop or
# knight = "Tượng cánh Vua", "Mã cánh Hậu"; break = "nước đột phá"; pawn majority = "đa số Tốt"; unbalanced =
# "bất đối xứng" ("mất cân bằng" would suggest one side is better); hypermodern = "siêu hiện đại"; the
# loanword "fianchetto" stays (Vietnamese chess books use it), beginner lines say "đưa Tượng ra đường chéo dài".
# Numbers that are not squares are written in words ("nước thứ hai", "hàng thứ sáu", "thế kỷ mười chín").

opening.variation.ck_advance.advanced = Trắng giành không gian bằng e5; Đen đưa Tượng ra f5 trước khi đi e6 và phản đòn bằng c5.
